module-info.java

The JAR is the module com.octopusfile. Packages marked @Internal are not exported.

java/**
 * OctopusFile: a file toolkit for Java 17+ with no external libraries.
 * <p>
 * Packages marked {@code @Internal} (support.*, infrastructure.concurrency,
 * .filesystem, .resources) are not exported: other modules cannot use them.
 * On the classpath (non-modular applications) the JAR works as before.
 * <p>
 * X25519 (sharing nearby) comes from the {@code jdk.crypto.ec} provider; in a
 * custom runtime add it with {@code jlink --add-modules jdk.crypto.ec}.
 */
module com.octopusfile {
    requires java.logging;
    requires java.net.http;   // WebDAV, internet downloads
    requires java.xml;        // WebDAV (PROPFIND answers)

    exports com.octopusfile;
    exports com.octopusfile.infrastructure.compression;
    exports com.octopusfile.infrastructure.errors;
    exports com.octopusfile.infrastructure.hashing;
    exports com.octopusfile.infrastructure.network;
    exports com.octopusfile.infrastructure.security;
    exports com.octopusfile.infrastructure.storage;
    exports com.octopusfile.modules.analysis;
    exports com.octopusfile.modules.backup;
    exports com.octopusfile.modules.duplicates;
    exports com.octopusfile.modules.filtering;
    exports com.octopusfile.modules.manipulation;
    exports com.octopusfile.modules.monitoring;
    exports com.octopusfile.modules.organization;
    exports com.octopusfile.modules.reading;
    exports com.octopusfile.modules.search;
    exports com.octopusfile.modules.sync;
    exports com.octopusfile.modules.transfer;
    exports com.octopusfile.modules.trash;
    exports com.octopusfile.modules.utils;
    exports com.octopusfile.modules.writing;
}
